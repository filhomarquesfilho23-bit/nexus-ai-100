async function chamarGemini(promptUsuario) { const !apiKey = localStorage.getItem('gemini_api_key'); if (!apiKey) { alert("Por favor, clique no ícone de chave verde no topo da tela e insira sua chave da API do Gemini."); return null; } const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3.8-flash:generateContent?key=${apiKey}`; const dadosRequisicao = { contents: [ { parts: [ { text: promptUsuario } ] } ] }; try { const resposta = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(dadosRequisicao) }); const resultado = await resposta.json(); if (resposta.ok) { return resultado.candidates[0].content.parts[0].text; } else { console.error("Erro na API:", resultado); alert("Erro na API do Gemini: " + (resultado.error?.message || "Erro desconhecido")); return null; } } catch (erro) { console.error("Erro de conexão:", erro); alert("Falha ao conectar com a API do Gemini."); return null; } }
